% Slide — Leverage Cycle Precedent
% ----------------------------
\begin{frame}{[DRAFT] Foundational Research}

\textbf{Endogenous leverage is central to financial instability.}

\vspace{0.3cm}

\begin{itemize}
    \item \textbf{Geanakoplos (2010)} formalizes the \emph{leverage cycle}:
    \begin{itemize}
        \item leverage, asset prices, and volatility are jointly determined
        \item falling margins inflate prices; tightening margins trigger fire sales
    \end{itemize}

    \vspace{0.2cm}

    \item Small shocks can generate large downturns through
    \begin{itemize}
        \item forced deleveraging
        \item collateral constraints
        \item feedback between prices and borrowing capacity
    \end{itemize}

    \vspace{0.2cm}

    \item The framework has shaped research on
    \begin{itemize}
        \item housing and mortgage markets
        \item repo and shadow banking
        \item margin-based securities lending
    \end{itemize}
\end{itemize}

\end{frame}


\begin{frame}{[DRAFT] DeFi as an Extension of the Leverage Cycle}

\textbf{DeFi lending protocols operationalize the leverage cycle mechanically.}

\vspace{0.3cm}

\begin{itemize}
    \item Collateral constraints are explicit and rule-based
    \begin{itemize}
        \item loan-to-value ratios
        \item automated liquidation thresholds
    \end{itemize}

    \vspace{0.2cm}

    \item Deleveraging occurs through \textbf{automatic, on-chain liquidations}
    \begin{itemize}
        \item no discretion, no negotiation
        \item immediate collateral sales
    \end{itemize}

    \vspace{0.2cm}

    \item All components of the leverage cycle are \textbf{observable}
    \begin{itemize}
        \item leverage (utilization, borrowed vs.\ supplied)
        \item liquidation flows
        \item collateral price volatility
    \end{itemize}

    \vspace{0.2cm}

    \item \textbf{Implication:}
    DeFi provides a laboratory to empirically test leverage-cycle dynamics
    across multiple financial systems in real time.
\end{itemize}

\end{frame}

% ----------------------------
% Slide — PVAR Framework
% ----------------------------
\begin{frame}{[DRAFT] Econometric Framework: Panel VAR (Pedroni)}

\textbf{The analysis uses a Pedroni-style Panel Vector Autoregression (PVAR).}

\vspace{0.3cm}

\begin{itemize}
    \item Developed to study \textbf{dynamic interactions} across heterogeneous units
    \item Combines:
    \begin{itemize}
        \item time-series dynamics (VAR)
        \item cross-sectional information (panel structure)
    \end{itemize}

    \vspace{0.2cm}

    \item Allows all variables to be \textbf{jointly endogenous}
    \begin{itemize}
        \item leverage
        \item liquidation activity
        \item collateral volatility
    \end{itemize}

    \vspace{0.2cm}

    \item Fixed effects capture \textbf{persistent protocol-level heterogeneity}
    \begin{itemize}
        \item risk parameters
        \item liquidation design
        \item asset composition
    \end{itemize}
\end{itemize}

\end{frame}

% ----------------------------
% Slide — Identification and Limits
% ----------------------------
\begin{frame}{[DRAFT] What the PVAR Can (and Cannot) Speak To}

\textbf{What this approach is well-suited to identify}

\vspace{0.2cm}

\begin{itemize}
    \item Dynamic responses to shocks via impulse response functions
    \item Direction and persistence of feedback effects
    \item Average behavior across protocols over time
    \item Evidence of amplification versus stabilization dynamics
\end{itemize}

\vspace{0.3cm}

\textbf{Key constraints of the methodology}

\vspace{0.2cm}

\begin{itemize}
    \item Reduced-form identification
    \item No structural interpretation without additional restrictions (explains how variables move together, not mechanism which drives those movements)
\end{itemize}


\end{frame}

% Slide — Utilization Ratio
% ----------------------------
\begin{frame}{[DRAFT] Leverage Measure: Utilization Ratio}

\textbf{Utilization captures aggregate leverage pressure within a lending pool.}

\vspace{0.3cm}

\[
\text{Utilization}_{t} =
\frac{\text{Total Borrowed}_{t}}{\text{Total Supplied}_{t}}
\]

\vspace{0.3cm}

\begin{itemize}
    \item Measures how aggressively liquidity is being borrowed
    \item Higher utilization implies:
    \begin{itemize}
        \item thinner liquidity buffers
        \item greater sensitivity to price shocks
    \end{itemize}

    \vspace{0.2cm}

    \item Interpreted as a \textbf{protocol-level analogue of leverage}
    \begin{itemize}
        \item reflects borrower risk appetite
        \item responds endogenously to prices and interest rates
    \end{itemize}

    \vspace{0.2cm}

\end{itemize}

\end{frame}

% ----------------------------
% Slide — Liquidations
% ----------------------------
\begin{frame}{[DRAFT] Liquidations: Forced Deleveraging}

\textbf{Liquidations measure realized deleveraging through collateral sales.}

\vspace{0.3cm}

\[
\text{Liquidation}_{t}
=
\sum \big(
\text{Collateral Liquidated}_{k,t}
\times
\text{Collateral Price}_{k,t}
\big)
\]

\vspace{0.3cm}

\begin{itemize}
    \item Captures the \textbf{intensive margin} of stress
    \begin{itemize}
        \item not just whether liquidations occur
        \item but how economically large they are
    \end{itemize}

    \vspace{0.2cm}

    \item Liquidations are aggregated daily at the protocol level
\end{itemize}

\end{frame}

% Slide — Volatility Measure
% ----------------------------
\begin{frame}{[DRAFT] Volatility: Collateral Basket Risk}

\textbf{Volatility is measured at the protocol level using collateral exposure.}

\vspace{0.2cm}

\textbf{Step 1: Log returns for each collateral asset}
\[
r_{i,t} = \ln(p_{i,t}) - \ln(p_{i,t-1})
\]

\vspace{0.2cm}

\textbf{Step 2: Lagged collateral weights}
\[
w_{i,t-1} =
\frac{\text{Collateral}_{i,t-1}}
{\sum_{j=1}^{N} \text{Collateral}_{j,t-1}}
\]

\vspace{0.2cm}

\textbf{Step 3: Protocol-level collateral return}
\[
r^{\text{protocol}}_{t}
=
\sum_{i=1}^{N} w_{i,t-1} \cdot r_{i,t}
\]

\vspace{0.2cm}

\textbf{Step 4: Rolling volatility}
\[
\sigma_{t}
=
\sqrt{\frac{1}{H}
\sum_{\tau=t-H+1}^{t}
\left(r^{\text{protocol}}_{\tau}
-
\bar{r}^{\text{protocol}}\right)^2}
\]

\end{frame}
